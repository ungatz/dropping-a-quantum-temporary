\newcommand{\stmtRecordWitness}{
\begin{theorem}[the criterion, one record at a time]\label{thm:perrecord}
Let $M$ be well typed with input context $\Omega$, and let $s$ drop a block of temporaries allocated in state $\psi$.
On a measurement record $o$ whose path includes $s$, the following four statements are equivalent.
\begin{enumerate}
\item $s$ is safe on $o$.
\item
  \begin{minipage}[t]{.59\linewidth}
  \raggedright
  There are a unit vector $g\in\Hv$ and a linear contraction
  $W_o:\den\Omega\to\Hrest$ such that:
  \end{minipage}\hfill
  \begin{minipage}[t]{.38\linewidth}
  $\forall x,\quad\begin{gathered}[t]
  \sigma_s^o x=g\otimes W_o x,\\
  \lVert W_o x\rVert\le\lVert x\rVert.
  \end{gathered}$
  \end{minipage}\par\medskip
\item
  Some witness built as above has kept-record map $\sigma_s^o$, after omitting the fixed $\ket0_a$ factor.
\item There is a witness as in (3) whose kept-record map, after omitting the fixed $\ket0_a$ factor,
  agrees with $\pfx{M}{s}$ on $o$ in every closed context. Measurement-free contexts suffice.
\end{enumerate}
\end{theorem}}
