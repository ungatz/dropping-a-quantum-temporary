% manuscript thm:meta, without the term-calculus clause
\newcommand{\stmtMeta}{%
\begin{theorem}[metatheory of the lifted calculus]\label{thm:meta}
Fix a measurement record $o$.
\begin{enumerate}
\item A well-typed configuration is terminal or steps; the step preserves
  typing, the parameter store binds exactly the parameters in scope, and no
  step names a dropped temporary.
\item Every run halts, and its terminal joint state is $E_o(c)$ applied to the
  visible input, where $E_o(c)$ is the selected denotation of $c$ along $o$.
\end{enumerate}
Hence the drop of $c$ is operationally safe under $o$ if and only if $E_o(c)$
is drop-safe.
\end{theorem}}

% the proportionality lemma (body sec 3.4; proof in A.1)
\newcommand{\stmtProp}{%
\begin{lemma}[proportionality]\label{lem:prop}
Let $K_0, \ldots, K_{d-1}$ and $M$ be linear maps on $\R^{m}$, and suppose
$\sum_i (K_i x)(K_i x)^{\top} = (Mx)(Mx)^{\top}$ for every $x$. Then $K_i =
c_i M$ for scalars $c_i$.
\end{lemma}}

% the summed-channel pair (body sec 3.4; proof in A.1)
\newcommand{\stmtChannel}{%
\begin{proposition}[the summed channel does not decide]\label{prop:channel}
There are two well-typed commands on a temporary qubit and a visible qubit,
each with two reachable records, that have the same summed channel $\Phi_0 +
\Phi_1$ and the same weight on every branch at every input, and opposite
verdicts on every branch.
\end{proposition}}

% product at every input: the constant-factor dichotomy (body sec 3.1; proof in A.1)
\newcommand{\stmtProductwise}{%
\begin{proposition}[a product at every input]\label{prop:productwise}
Let $K : \R^{m} \to \R^{d} \otimes \R^{m}$ be linear, and suppose $Kv$ is
a product vector, or zero, for every $v$. Then $K$ has one of exactly two
shapes:
\begin{description}
\item[the temporary is fixed] $K = g \otimes W$ for a vector $g \in \R^{d}$
  and a matrix $W$ on $\R^{m}$ --- the visible register varies, the temporary
  does not;
\item[the visible register is fixed] $Kv = Gv \otimes w$ for a fixed nonzero
  $w \in \R^{m}$ and a linear $G : \R^{m} \to \R^{d}$ --- the temporary
  varies, the visible register does not.
\end{description}
The first shape is drop safety. A map of the second is drop-safe exactly when
$G$ has rank at most one, so such an isometry with $m \ge 2$ never is.
\end{proposition}}

% the relativised criterion (body sec 5; proof in A.2)
\newcommand{\stmtRelativised}{%
\begin{theorem}[the relativised criterion]\label{thm:relativised}
Let $\mathcal A$ be a gate ring completable in the temporary's dimension
and $U$ a program over $\mathcal A$. Then $U$ is weakenable by gates over
$\mathcal A$ if and only if $U$ is drop-safe and $\mathcal A$ splits
$\mathrm{proj}(U)$.
\end{theorem}}

% without typing (body sec 4; proof in A.1)
\newcommand{\stmtUntyped}{%
\begin{proposition}[without typing]\label{prop:untyped}
For a program of the grammar whose gates are arbitrary linear maps,
weakenability is equivalent to drop safety together with $\|Kv\| = \|v\|$ for
every $v$. The norm clause cannot be dropped: at $d = m = 1$ the one-gate
program that doubles the visible register is drop-safe and not weakenable.
\end{proposition}}


% what the Gram matrix is (body sec 3.1; proof in A.1)
\newcommand{\stmtGram}{%
\begin{proposition}[what the Gram matrix is]\label{prop:gram}
Let $K$ be the denotation of a typed program, with Kraus operators $K_i$
and Gram matrix $\Gamma$.
\begin{enumerate}
\item The drop is safe if and only if $K_i^{\dagger} K_j = c_{ij} I$ for
  all $i, j$, and then $c = \Gamma/m$; over $\C$, if and only if the
  temporary's final state is the same at every unit input.
\item $\operatorname{rank}\Gamma$ is the least temporary dimension of a
  typed program whose visible channel $\rho \mapsto \sum_i K_i \rho K_i^{\dagger}$
  is that of $K$.
\item $\Gamma$ is unchanged by a visible gate after the scope, and a gate
  $U$ on the temporary carries it to $\bar U \Gamma U^{\top}$.
\end{enumerate}
\end{proposition}}


\newcommand{\stmtClassicalCost}{%
\begin{theorem}[the classical price]\label{thm:classicalcost}
Deciding whether a circuit program over the gate set $\{X, \mathrm{CNOT},
\mathrm{Toffoli}\}$ is drop-safe is coNP-complete.
\end{theorem}}

\newcommand{\stmtAverageCost}{%
\begin{proposition}[the price at the maximally mixed input]\label{prop:avgcost}
Fix a polynomial $p : \N \to \N$ with $p(n) \ge 1$ for every $n$, let $\varepsilon = 1/p(\text{size})$, and let the
circuit program's gates come from a fixed finite set. Deciding, for such
a program promised either drop-safe or $\bar\iota(K) \ge \varepsilon$,
which holds is in BQP.
\end{proposition}}


% contextual indistinguishability (body sec 4.1; proof in A.1)
\newcommand{\stmtCtxEquiv}{%
\begin{theorem}[the criterion is contextual indistinguishability]\label{thm:ctxequiv}
Let $F$ be any class of visible gates containing the reflections. A typed
program is weakenable if and only if $F$ does not distinguish it from some
separable program.
\end{theorem}}

% the verdict is outcome-dependent (body sec 6; proof in A.3)
\newcommand{\stmtNoText}{%
\begin{theorem}[the verdict is outcome-dependent]\label{thm:notext}
Let $c$ be the command above.
\begin{enumerate}
\item $c$ is well typed, with an empty parameter context and a live
  temporary.
\item Under the record selecting $\{0,1\}$ the selected denotation is
  $K_{\mathrm{in}} v = \tfrac23({\color{accent}|0\rangle \otimes v} + {\color{accentwarm}|1\rangle \otimes Xv})$
  and the drop is unsafe; under the record selecting $\{2\}$ it is
  $K_{\mathrm{out}} v = \tfrac13\, {\color{accentplum}|2\rangle \otimes X^{2} v}$ and the drop is
  safe. Both records are reachable, with weights $8/9$ and $1/9$ at every unit
  input.
\item No lift-free command has a denotation agreeing with $c$ under both
  records.
\end{enumerate}
\end{theorem}}

% manuscript thm:cliffordT; part (2) at any visible register
\newcommand{\stmtCliffordT}{%
\begin{theorem}[the Clifford$+T$ coefficient ring splits]\label{thm:cliffordT}
Let $\mathcal R = \Z[1/\sqrt2, i]$ and $d \ge 1$.
\begin{enumerate}
\item Every rank-one projection on $\C^{d}$ with entries in $\mathcal R$ is
  $zz^{\dagger}$ for a vector $z$ over $\mathcal R$.
\item Consequently, at a temporary of any dimension and any visible register,
  a program over $\mathcal R$ is weakenable by gates over $\mathcal R$ if and
  only if it is drop-safe.
\end{enumerate}
\end{theorem}}

% the four restricted rings of Amy--Glaudell--Ross (proof in A.2)
\newcommand{\stmtRings}{%
\begin{proposition}[the restricted rings]\label{prop:rings}
Over $\Z[1/2, i]$ and over $\Z[i/\sqrt2]$, every rank-one projection with
entries in the ring, in every dimension, is $zz^{\dagger}$ for a $z$ over the
ring. Over $\Z[1/2]$ the projection onto $|{+}\rangle$ is not, and over
$\Z[1/\sqrt2]$ the projection onto $\cos\tfrac{\pi}{8}|0\rangle +
\sin\tfrac{\pi}{8}|1\rangle$ is not.
\end{proposition}}


% manuscript thm:bundle, restructured; (2) is the manuscript's lemma
\newcommand{\stmtBundle}{%
\begin{theorem}[the composite temporary]\label{thm:bundle}
Let $c$ be a well-typed lifted command over temporaries $t_1, \ldots, t_n$
with $n \ge 1$, and $\langle c \rangle$ its composite temporary.
\begin{enumerate}
\item $\langle c \rangle$ is well typed and, along every measurement record, runs
  with the same parameter store and the same joint state as $c$.
\item Let $c$ be measurement-free and well-scoped. For every $k$ such that $c$
  contains $\drop\ t_k$, dropping $t_k$ at that site is safe if and only if
  $t_k$ is separately constant in the final state.
\item Let $c$ be measurement-free. Every temporary is separately constant
  in the final state if and only if the composite drop is safe and its final state $g$ is a full product
  $g_1 \otimes \cdots \otimes g_n$; at $n = 2$, if and only if the composite drop
  is safe and the $d_1 \times d_2$ matrix $g$ has vanishing $2 \times 2$ minors.
\end{enumerate}
\end{theorem}}

% manuscript thm:nested
\newcommand{\stmtTrichotomy}{%
\begin{proposition}[the nested witness: a trichotomy]\label{prop:trichotomy}
Let $P$ be well-typed, measurement-free and well-scoped, with temporaries
$t_1, \ldots, t_n$, a nonempty visible register, and gates over a gate ring
$\mathcal A$. Exactly one of the following holds.
\begin{enumerate}
\item The composite drop is unsafe. No witness of any shape exists.
\item The composite drop is safe and its final state $g$ is not a full product.
  No nested witness exists.
\item The composite drop is safe and $g = g_1 \otimes \cdots \otimes g_n$. Every
  temporary is separately constant; a nested witness over $\mathcal A$ exists
  \emph{only if} each ray $[g_k]$ carries a unit vector with entries in
  $\mathcal A$, and \emph{does} exist if each $[g_k]$ carries a real unit
  vector $u_k$ over $\mathcal A$ with $\|e_{i_k} - u_k\|^{2}$ zero or
  invertible in $\mathcal A$.
\end{enumerate}
\end{proposition}}

% manuscript thm:typingsplit
\newcommand{\stmtTypingSplit}{%
\begin{theorem}[typing decomposes per lifetime]\label{thm:typingsplit}
A lift-free multi-temporary command is well typed if and only if every gate
is a unitary of its declared shape and, for every name $t$, the projection of
the command to $t$ obeys the one-temporary discipline. The judgement carries
no cross-temporary information.
\end{theorem}}

% manuscript thm:perwire, with the pair inside the statement (proof in A.4)
\newcommand{\stmtPerwire}{%
\begin{theorem}[no per-lifetime analysis decides drop safety]\label{thm:perwire}
Take two temporary qubits $a, b$ and one visible qubit $v$, all allocated at
$|0\rangle$, and the programs
\[
\begin{array}{l@{\qquad}l}
P_{\mathrm{safe}}: & H_b;\; \mathrm{CX}(v{\to}a);\; \mathrm{CX}(v{\to}a);\; {\color{accentwarm}\mathrm{CX}(b{\to}v)}\\
P_{\mathrm{unsafe}}: & H_b;\; \mathrm{CX}(v{\to}a);\; {\color{accentwarm}\mathrm{CX}(b{\to}v)};\; \mathrm{CX}(v{\to}a).
\end{array}
\]
Their projections to $a$ and to $b$ are equal, dropping $a$ is safe after
$P_{\mathrm{safe}}$ and unsafe after $P_{\mathrm{unsafe}}$. Hence no
predicate on families of per-temporary projections agrees with drop safety
on all well-typed straight-line two-temporary programs.
\end{theorem}}
